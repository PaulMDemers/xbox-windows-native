# Experimental: name/structure audit target, not yet boot compatible.
if(NOT ARCH STREQUAL "i386" OR NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "XBOX_NT5_HAL requires an i386 Release (free UP) build")
endif()
remove_definitions(-D_WIN32_WINNT=0x502 -DWINVER=0x502)
add_definitions(-D_WIN32_WINNT=0x500 -DWINVER=0x500 -DNTDDI_VERSION=0x05000000
                -DXBOX_NT5_HAL -DSARCH_XBOX)
include_directories(${CMAKE_CURRENT_BINARY_DIR}/..)
set(NT5_SOURCE)
foreach(source ${HAL_XBOX_SOURCE} ${HAL_XBOX_ASM_SOURCE} ${HAL_UP_SOURCE})
    if(IS_ABSOLUTE "${source}")
        set_source_files_properties("${source}" PROPERTIES GENERATED TRUE)
        list(APPEND NT5_SOURCE "${source}")
    else()
        list(APPEND NT5_SOURCE "../${source}")
    endif()
endforeach()
spec2def(halxboxnt5.dll halnt5.spec)
add_asm_files(nt5_boundary_asm boundary.S)
add_library(halxboxnt5 MODULE ${NT5_SOURCE} ${nt5_boundary_asm} fmutex.c trapentry.c kdport.c translate.c rootbus.c entryprobe.c mapprobe.c ../xbox/halxbox.rc
            ${CMAKE_CURRENT_BINARY_DIR}/halxboxnt5.def)
target_link_libraries(halxboxnt5 cportlib libcntpr setjmp fast486 ${PSEH_LIB})
add_importlibs(halxboxnt5 ntoskrnl)
add_dependencies(halxboxnt5 psdk asm bugcodes xdk xbox_nt5_pcidata)
set_module_type(halxboxnt5 kerneldll ENTRYPOINT HalInitSystem 8)
if(NOT MSVC)
    target_link_libraries(halxboxnt5 -lgcc)
    target_link_options(halxboxnt5 PRIVATE -Wl,--subsystem,native:5.0
        -Wl,--minor-image-version,0 -Wl,--minor-os-version,0)
endif()
