remove_definitions(-D_WIN32_WINNT=0x502 -DWINVER=0x502)
add_definitions(-D_WIN32_WINNT=0x500 -DWINVER=0x500 -DNTDDI_VERSION=0x05000000)
add_library(xboxvnt5 MODULE xboxvnt5.c ../xboxvmp.rc)
set_module_type(xboxvnt5 kernelmodedriver)
add_importlibs(xboxvnt5 videoprt)
target_link_libraries(xboxvnt5 libcntpr)
if(NOT MSVC)
    target_link_options(xboxvnt5 PRIVATE -Wl,--subsystem,native:5.0
        -Wl,--minor-image-version,0 -Wl,--minor-os-version,0)
endif()

option(XBOX_NT5_VIDEO_DIAGNOSTIC "Stop at first failing NT5 video initialization stage" OFF)
if(XBOX_NT5_VIDEO_DIAGNOSTIC)
    target_compile_definitions(xboxvnt5 PRIVATE XBOX_NT5_VIDEO_DIAGNOSTIC=1)
    add_importlibs(xboxvnt5 ntoskrnl)
endif()

option(XBOX_NT5_VIDEO_ENTRY_DIAGNOSTIC "Stop before NT5 video DriverEntry body" OFF)
if(XBOX_NT5_VIDEO_ENTRY_DIAGNOSTIC)
    if(NOT XBOX_NT5_VIDEO_DIAGNOSTIC)
        message(FATAL_ERROR "Video entry diagnostic requires video failure diagnostic")
    endif()
    target_compile_definitions(xboxvnt5 PRIVATE XBOX_NT5_VIDEO_ENTRY_DIAGNOSTIC=1)
endif()

option(XBOX_NT5_VIDEO_ADAPTER_DIAGNOSTIC "Stop after NT5 video resource discovery" OFF)
if(XBOX_NT5_VIDEO_ADAPTER_DIAGNOSTIC)
    if(NOT XBOX_NT5_VIDEO_DIAGNOSTIC OR XBOX_NT5_VIDEO_ENTRY_DIAGNOSTIC)
        message(FATAL_ERROR "Adapter diagnostic requires failure checks and entry checkpoint OFF")
    endif()
    target_compile_definitions(xboxvnt5 PRIVATE XBOX_NT5_VIDEO_ADAPTER_DIAGNOSTIC=1)
endif()

option(XBOX_NT5_VIDEO_PNP_DIAGNOSTIC "Stop before NT5 video PnP start" OFF)
if(XBOX_NT5_VIDEO_PNP_DIAGNOSTIC)
    if(NOT XBOX_NT5_VIDEO_DIAGNOSTIC OR XBOX_NT5_VIDEO_ENTRY_DIAGNOSTIC OR XBOX_NT5_VIDEO_ADAPTER_DIAGNOSTIC)
        message(FATAL_ERROR "PnP diagnostic requires failure checks and other checkpoints OFF")
    endif()
    target_compile_definitions(xboxvnt5 PRIVATE XBOX_NT5_VIDEO_PNP_DIAGNOSTIC=1)
    target_sources(xboxvnt5 PRIVATE video_pnp.c)
endif()

option(XBOX_NT5_VIDEO_TRACE "Trace NT5 video PnP to an existing seeded file" OFF)
if(XBOX_NT5_VIDEO_TRACE)
    if(NOT XBOX_NT5_VIDEO_PNP_DIAGNOSTIC)
        message(FATAL_ERROR "Video trace requires PnP diagnostic wrappers")
    endif()
    target_compile_definitions(xboxvnt5 PRIVATE XBOX_NT5_VIDEO_TRACE=1)
    option(XBOX_NT5_VIDEO_TRACE_REUSE "Reuse a valid trace file across NT5 boots" OFF)
    if(XBOX_NT5_VIDEO_TRACE_REUSE)
        target_compile_definitions(xboxvnt5 PRIVATE XBOX_NT5_VIDEO_TRACE_REUSE=1)
    endif()
    target_sources(xboxvnt5 PRIVATE video_trace.c)
    add_importlibs(xboxvnt5 hal)
endif()

option(XBOX_NT5_VIDEO_MAPPING_AUDIT "Record mapping/PAT/MTRR state without changing it" OFF)
if(XBOX_NT5_VIDEO_MAPPING_AUDIT)
    if(NOT XBOX_NT5_VIDEO_TRACE)
        message(FATAL_ERROR "Mapping audit requires the existing trace")
    endif()
    target_compile_definitions(xboxvnt5 PRIVATE XBOX_NT5_VIDEO_MAPPING_AUDIT=1)
    target_sources(xboxvnt5 PRIVATE video_mapping_audit.c)
endif()

option(XBOX_NT5_VIDEO_DIRECT_RAM "Map the reserved framebuffer RAM directly without changing cache flags" OFF)
if(XBOX_NT5_VIDEO_DIRECT_RAM)
    target_compile_definitions(xboxvnt5 PRIVATE XBOX_NT5_VIDEO_DIRECT_RAM=1)
endif()

option(XBOX_NT5_VIDEO_DIRECT_MAP "Own a noncached reserved-RAM mapping outside PCI translation" OFF)
if(XBOX_NT5_VIDEO_DIRECT_MAP)
    if(NOT XBOX_NT5_VIDEO_DIRECT_RAM OR NOT XBOX_NT5_VIDEO_TRACE)
        message(FATAL_ERROR "Direct mapping requires direct-RAM geometry and the unload wrapper")
    endif()
    target_compile_definitions(xboxvnt5 PRIVATE XBOX_NT5_VIDEO_DIRECT_MAP=1)
    target_sources(xboxvnt5 PRIVATE video_direct_map.c)
    add_importlibs(xboxvnt5 ntoskrnl hal)
endif()
